## Exercice 2 : Calcul de prix TTC avec remise

DEBUT
  VARIABLE prixHT, prixTTC, prixFinal : REEL
  VARIABLE tauxTVA, montantTVA : REEL
  VARIABLE pourcentageRemise, montantRemise : REEL

  ECRIRE("Prix HT de l'article : ")
  LIRE(prixHT)
  ECRIRE("Taux de TVA en % : ")
  LIRE(tauxTVA)
  ECRIRE("Pourcentage de remise : ")
  LIRE(pourcentageRemise)

  montantTVA <- prixHT * tauxTVA /100
  prixTTC <- prixHT + montantTVA
  montantRemise <- prixTTC * pourcentageRemise / 100
  prixFinal <- prixTTC - montantRemise

  ECRIRE("Prix HT : ", prixHT)
  ECRIRE("TVA (%) : ", tauxTVA)
  ECRIRE("Remise (%) : ", pourcentageRemise)
  ECRIRE("Montant TVA : ", montantTVA)
  ECRIRE("Prix TTC : ", prixTTC)
  ECRIRE("Montant remise : ", montantRemise)
  ECRIRE("Prix final : ", prixFinal)
FIN
