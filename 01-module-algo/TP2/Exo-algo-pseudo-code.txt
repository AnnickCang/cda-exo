## Exercice 1 : Calculatrice simple

DEBUT
  VARIABLE nombre1, nombre2, resultat : REEL
  VARIABLE operateur : CHAINE

  ECRIRE("Entrez le 1er nombre : ")
  LIRE(nombre1)
  ECRIRE("Entrez le 2nd nombre : ")
  LIRE(nombre2)
  ECRIRE("Opération (+, -, *, /) : ")
  LIRE(operateur)

  TESTER(operateur)
    CAS("+")
      resultat <- nombre1 + nombre2
      ECRIRE(nombre1, " + ", nombre2, " = ", resultat)
    CAS("-")
      resultat <- nombre1 - nombre2
      ECRIRE(nombre1, " - ", nombre2, " = ", resultat)
    CAS("*")
      resultat <- nombre1 * nombre2
      ECRIRE(nombre1, " * ", nombre2, " = ", resultat)
    CAS("/")
      SI nombre2 == 0
        ECRIRE("Erreur : division par zéro")
      SINON
        resultat <- nombre1 / nombre2
        ECRIRE(nombre1, " / ", nombre2, " = ", resultat)
      FIN SI
    AUTRE
      ECRIRE("Erreur : opérateur inconnu")
  FIN TESTER
FIN


## Exercice 2 : Jeu "Plus grand / Plus petit"

DEBUT
  VARIABLE nombreMystere, nombre, nombreEssais : ENTIER
  VARIABLE gagne : BOOLEEN

  gagne <- FAUX
  nombreEssais <- 0
  // nombre aléatoire entr 1 et 100 choisi par l'ordinateur
  nombreMystere <- ALEATOIRE(1, 100)

  ECRIRE("Jeu : plus grand - plus petit")
  ECRIRE("Devinez un nombre entre 1 et 100 en un minimum d'essais")

  REPETER
    nombreEssais <- nombreEssais + 1
    ECRIRE("Nombre : ")
    LIRE(nombre)

    SI (nombre < 1 OU nombre > 100)
      ECRIRE("Le nombre doit être compris entre 1 et 100 inclus")
      nombreEssais <- nombreEssais - 1
    SINON SI (nombre == nombreMystere)
      gagne <- VRAI
      ECRIRE("Bravo ! Trouvé en ", nombreEssais, " essais")
    SINON SI (nombre < nombreMystere)
      ECRIRE("Plus grand !")
    SINON
      ECRIRE("Plus petit !")
    FIN SI
  TANT QUE (gagne == FAUX)
FIN

Pattern d'algorithme utilisé : sentinelle
Stratégie optimale pour minimiser le nombre d'essais : dichotomie
Nombre maximum d'essais pour trouver un nombre entre 1 et 100 avec la dichotomie : 7


## Exercice 3 : FizzBuzz amélioré

DEBUT
  FONCTION getFizzbuzzResult(nombre : ENTIER) : CHAINE
  DEBUT
    VARIABLE texte : CHAINE

    texte <- ""

    SI ((nombre MOD 3) = 0)
      texte <- texte + "Fizz"
    FIN SI
    SI ((nombre MOD 5) = 0)
      texte <- texte + "Buzz"
    FIN SI
    SI ((nombre MOD 7) = 0)
      texte <- texte + "Wazz"
    FIN SI
    SI (texte = "")
      texte <- (CHAINE) nombre // nombre est casté en CHAINE
    FIN SI

    RETOURNER texte
  FIN

  VARIABLE i : ENTIER
  VARIABLE texte : CHAINE
  
  texte <- ""

  POUR i ALLANT DE 1 A 21 FAIRE
    texte <- textte + getFizzbuzzResult(i)
  FIN POUR
  ECRIRE(texte)
FIN


## Exercice 4 : Validation de mot de passe

// Pour les fonctions ci-dessous, je considère que je réutiliserai une fonction standard (si elle existe, ou une autre méthode) 
// de JavaScript ou Java qui retourne un booléen pour vérifier la condition demandée

DEBUT FONCTION aLaTailleMinimale(motDePasse : CHAINE) : BOOLEEN
  RETOURNER (LONGUEUR(motDePasse) >= 8)
FIN FONCTION

DEBUT FONCTION contientMajuscule(motDePasse : CHAINE) : BOOLEEN
  RETOURNER (MAJUSCULE(motDePasse))
FIN FONCTION

DEBUT FONCTION contientMinuscule(motDePasse : CHAINE) : BOOLEEN
  RETOURNER (MINUSCULE(motDePasse))
FIN FONCTION

DEBUT FONCTION contientChiffre(motDePasse : CHAINE) : BOOLEEN
  RETOURNER (CHIFFRE(motDePasse))
FIN FONCTION

DEBUT
  VARIABLE motDePasse : CHAINE
  VARIABLE valide : BOOLEEN

  valide <- VRAI

  ECRIRE("Mot de passe : ")
  LIRE(motDePasse)

  SI aLaTailleMinimale(motDePasse)
    ECRIRE("Longueur >= 8 : V")
  SINON
    valide <- FAUX
    ECRIRE("Longueur >= 8 : X")
  FIN SI

  SI contientMajuscule(motDePasse)
    ECRIRE("Majuscule : V")
  SINON
    valide <- FAUX
    ECRIRE("Majuscule : X")
  FIN SI

  SI contientMinuscule(motDePasse)
    ECRIRE("Minsucule : V")
  SINON
    valide <- FAUX
    ECRIRE("Minsucule : X")
  FIN SI

  SI contientChiffre(motDePasse)
    ECRIRE("Chiffre : V")
  SINON
    valide <- FAUX
    ECRIRE("Chiffre : X")
  FIN SI

  SI valide
    ECRIRE("Valide ? V")
  SINON
    ECRIRE("Valide ? X")
  FIN SI
FIN
